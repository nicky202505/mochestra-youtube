<script>
  const $ = (id) => document.getElementById(id);
  const resultsEl = $('results');
  const favListEl = $('favList');
  const recListEl = $('recList');
  const playerEl = $('player');
  const playBtn = $('playBtn');
  const keyDialog = $('keyDialog');
  const tabsEl = $('tabs');

  const STAR_SVG = '<svg viewBox="0 0 24 24"><polygon points="12 2 15.09 8.26 22 9.27 17 14.14 18.18 21.02 12 17.77 5.82 21.02 7 14.14 2 9.27 8.91 8.26"/></svg>';

  let items = [];          // 검색 결과
  let recommended = [];    // 추천 리스트 (준비 중)
  let selectedId = null;
  let activeTab = 'search';

  const store = {
    get(k) { try { return localStorage.getItem(k) || ''; } catch { return ''; } },
    set(k, v) {
      try {
        localStorage.setItem(k, v);
        return localStorage.getItem(k) === v;
      } catch {
        return false;
      }
    },
  };

  function loadFavorites() {
    try { return JSON.parse(store.get('yt_favorites') || '[]'); } catch { return []; }
  }
  function saveFavorites() { store.set('yt_favorites', JSON.stringify(favorites)); }
  let favorites = loadFavorites();
  function isFav(id) { return favorites.some(f => f.id === id); }
  function toggleFavorite(item) {
    if (isFav(item.id)) favorites = favorites.filter(f => f.id !== item.id);
    else favorites.push(item);
    saveFavorites();
    renderResults();
    if (activeTab === 'favorites') renderFavorites();
  }

  // 링크나 11자리 영상 ID면 바로 재생
  function parseVideoId(text) {
    const s = text.trim();
    if (/^[\w-]{11}$/.test(s)) return s;
    const m = s.match(/(?:youtu\.be\/|v=|\/embed\/|\/shorts\/|\/live\/)([\w-]{11})/);
    return m ? m[1] : null;
  }

  function play(id) {
    selectedId = id;
    playerEl.innerHTML = '';
    const f = document.createElement('iframe');
    f.src = `https://www.youtube.com/embed/${id}?autoplay=1&rel=0`;
    f.allow = 'autoplay; encrypted-media; picture-in-picture; fullscreen';
    f.allowFullscreen = true;
    f.referrerPolicy = 'strict-origin-when-cross-origin';
    playerEl.appendChild(f);
    document.querySelectorAll('.list-item').forEach(li =>
      li.classList.toggle('active-item', li.dataset.id === id));
  }

  function renderEmpty(el, text) {
    el.innerHTML = `<div class="empty"></div>`;
    el.firstChild.textContent = text;
  }

  function renderResults() {
    if (!items.length) return renderEmpty(resultsEl, '검색결과');
    resultsEl.innerHTML = '';
    for (const it of items) {
      const row = document.createElement('div');
      row.className = 'list-item';
      row.dataset.id = it.id;
      row.classList.toggle('active-item', it.id === selectedId);
      row.innerHTML = `<div class="title"></div><button type="button" class="star-btn" aria-label="즐겨찾기">${STAR_SVG}</button>`;
      row.querySelector('.title').textContent = it.title;
      const starBtn = row.querySelector('.star-btn');
      starBtn.classList.toggle('is-fav', isFav(it.id));
      starBtn.onclick = (e) => { e.stopPropagation(); toggleFavorite(it); };
      row.onclick = () => play(it.id);
      resultsEl.appendChild(row);
    }
  }

  function renderFavorites() {
    if (!favorites.length) return renderEmpty(favListEl, '즐겨찾기한 영상이 없어요');
    favListEl.innerHTML = '';
    for (const it of favorites) {
      const row = document.createElement('div');
      row.className = 'list-item';
      row.dataset.id = it.id;
      row.classList.toggle('active-item', it.id === selectedId);
      row.innerHTML = `<div class="title"></div>`;
      row.querySelector('.title').textContent = it.title;
      row.onclick = () => play(it.id);
      favListEl.appendChild(row);
    }
  }

  function renderRecommended() {
    if (!recommended.length) return renderEmpty(recListEl, '추천 영상이 없어요');
    recListEl.innerHTML = '';
    for (const it of recommended) {
      const row = document.createElement('div');
      row.className = 'list-item';
      row.dataset.id = it.id;
      row.classList.toggle('active-item', it.id === selectedId);
      row.innerHTML = `<div class="title"></div>`;
      row.querySelector('.title').textContent = it.title;
      row.onclick = () => play(it.id);
      recListEl.appendChild(row);
    }
  }

  function switchTab(tab) {
    activeTab = tab;
    for (const btn of tabsEl.querySelectorAll('.tab-btn')) {
      const on = btn.dataset.tab === tab;
      btn.classList.toggle('active', on);
      btn.setAttribute('aria-selected', String(on));
    }
    for (const panel of document.querySelectorAll('.panel')) {
      panel.classList.toggle('active', panel.id === `panel-${tab}`);
    }
    if (tab === 'favorites') renderFavorites();
    if (tab === 'recommended') renderRecommended();
  }

  tabsEl.addEventListener('click', (e) => {
    const btn = e.target.closest('.tab-btn');
    if (btn) switchTab(btn.dataset.tab);
  });

  const decode = (s) => { const t = document.createElement('textarea'); t.innerHTML = s; return t.value; };

  async function search(q) {
    const key = store.get('yt_api_key');
    if (!key) { keyDialog.showModal(); return; }
    renderEmpty(resultsEl, '검색 중…');
    try {
      const url = new URL('https://www.googleapis.com/youtube/v3/search');
      url.search = new URLSearchParams({
        part: 'snippet', type: 'video', maxResults: '10', q, key, regionCode: 'KR',
      });
      const res = await fetch(url);
      const data = await res.json();
      if (data.error) throw new Error(data.error.message);
      items = data.items.map(v => ({
        id: v.id.videoId,
        title: decode(v.snippet.title),
        channel: decode(v.snippet.channelTitle),
        thumb: v.snippet.thumbnails.medium?.url || v.snippet.thumbnails.default.url,
      }));
      renderResults();
    } catch (e) {
      const message = e.message.includes('Requests from referer')
        ? '검색 실패: Google Cloud에서 localhost 주소를 API 키 허용 리퍼러에 추가해 주세요.'
        : '검색 실패: ' + e.message;
      renderEmpty(resultsEl, message);
    }
  }

  $('searchForm').addEventListener('submit', (e) => {
    e.preventDefault();
    const q = $('q').value.trim();
    if (!q) return;
    const id = parseVideoId(q);
    if (id) { items = []; renderEmpty(resultsEl, '링크에서 바로 재생 중'); play(id); }
    else search(q);
  });

  // 한글 조합 중 Enter는 무시하고, 조합이 끝난 Enter만 검색
  $('q').addEventListener('keydown', (e) => {
    if (e.key !== 'Enter' || e.isComposing || e.keyCode === 229) return;
    e.preventDefault();
    $('searchForm').requestSubmit();
  });

  playBtn.addEventListener('click', () => {
    if (items.length) play(items[0].id);
    else $('q').focus();
  });

  $('keyLink').onclick = () => { $('keyInput').value = store.get('yt_api_key'); keyDialog.showModal(); };
  keyDialog.addEventListener('close', () => {
    if (keyDialog.returnValue !== 'save') return;
    const key = $('keyInput').value.trim();
    if (!store.set('yt_api_key', key)) {
      alert('API 키를 저장하지 못했어요. 브라우저의 사이트 데이터 차단 설정을 확인해 주세요.');
      return;
    }
    const q = $('q').value.trim();
    if (q && !parseVideoId(q)) search(q);
  });

  renderFavorites();
</script>
